public class Main {
    public static boolean leapYear(int y)
    {
        if(y%4==0)
        {
            if(y%100==0)
                if(y%400==0)
                    return true;
            if(y%4==0 && y%100 !=0) return true;
        }
        return false;
    }
    public static void main(String[] args)
    {
        for(int i=1;i<=100;i++)
            if(leapYear(i))
                System.out.println(i);
    }
}
